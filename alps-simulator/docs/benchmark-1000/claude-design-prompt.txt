  const dz=Math.max(0,Math.min(top(box),top(other))-Math.max(p.z,other.position.z));if(dz<=EPS)continue;
  let length=0;
  if(Math.abs(p.x+s.w-other.position.x)<EPS||Math.abs(other.position.x+other.size.w-p.x)<EPS)length+=Math.max(0,Math.min(p.y+s.d,other.position.y+other.size.d)-Math.max(p.y,other.position.y));
  if(Math.abs(p.y+s.d-other.position.y)<EPS||Math.abs(other.position.y+other.size.d-p.y)<EPS)length+=Math.max(0,Math.min(p.x+s.w,other.position.x+other.size.w)-Math.max(p.x,other.position.x));
  contact+=length*dz;if(Math.abs(top(other)-top(box))<EPS)sameTop+=length;
 }
 return {contact,sameTop};
}
/** Finite side-strip heuristic: narrow gaps between this box and an adjacent side
 * obstacle/wall, compared with the smallest known remaining footprint width/depth.
 * Corner strips can overlap. This is a tie-breaking proxy, not inaccessible area.
 */
function narrowSideStrips(box:Placement,input:PlanningInput){
 const remaining=input.types.filter(t=>(input.remaining[t.id]??0)>0);
 if(!remaining.length)return 0;
 const sizes=remaining.flatMap(t=>allowedOrientations(t.size,t.orientations,t.handling,t.maxLoadByAxis).map(o=>oriented(t.size,o)));
 const minX=Math.min(...sizes.map(s=>s.w)),minY=Math.min(...sizes.map(s=>s.d));
 let total=0;
 for(const axis of ['x','y'] as const){
  const along=axis==='x'?'y':'x',width=axis==='x'?'w':'d',length=axis==='x'?'d':'w',limit=axis==='x'?input.pallet.width:input.pallet.depth,min=axis==='x'?minX:minY;
  const start=box.position[along],end=start+box.size[length];
  const neighbors=input.placements.filter(p=>top(p)>box.position.z+EPS&&p.position.z<top(box)-EPS&&p.position[along]<end-EPS&&p.position[along]+p.size[length]>start+EPS);
  const edges=[...new Set([start,end,...neighbors.flatMap(p=>[Math.max(start,p.position[along]),Math.min(end,p.position[along]+p.size[length])])])].sort((a,b)=>a-b);
  for(let i=1;i<edges.length;i++){
   const mid=(edges[i]+edges[i-1])/2,cover=neighbors.filter(p=>mid>=p.position[along]&&mid<p.position[along]+p.size[length]);
   let before=box.position[axis],after=limit-box.position[axis]-box.size[width];
   for(const p of cover){const right=p.position[axis]+p.size[width];if(right<=box.position[axis]+EPS)before=Math.min(before,box.position[axis]-right);if(p.position[axis]>=box.position[axis]+box.size[width]-EPS)after=Math.min(after,p.position[axis]-box.position[axis]-box.size[width]);}
   for(const gap of [before,after])if(gap>EPS&&gap<min-EPS)total+=gap*(edges[i]-edges[i-1]);
  }
 }
 return total;
}
export function denseRank(box:Placement,input:PlanningInput):number[]{
 const stack=[...input.placements,box],mass=stack.reduce((s,p)=>s+p.weight,0);
 const cx=stack.reduce((s,p)=>s+p.weight*(p.position.x+p.size.w/2),0)/mass,cy=stack.reduce((s,p)=>s+p.weight*(p.position.y+p.size.d/2),0)/mass,cz=stack.reduce((s,p)=>s+p.weight*(p.position.z+p.size.h/2),0)/mass;
 const bounds=(Math.max(...stack.map(p=>p.position.x+p.size.w))-Math.min(...stack.map(p=>p.position.x)))*(Math.max(...stack.map(p=>p.position.y+p.size.d))-Math.min(...stack.map(p=>p.position.y)));
 const contacts=faceContacts(box,input);
 return [box.position.z,newColumnVoid(box,input.placements),narrowSideStrips(box,input),-contacts.sameTop,-contacts.contact,bounds,Math.max(...stack.map(top)),Math.hypot(cx-input.pallet.width/2,cy-input.pallet.depth/2),cz,box.position.y,box.position.x,box.orientation===0?0:1];
}
export function compareDenseRank(a:readonly number[],b:readonly number[]):number{
 for(let i=0;i<a.length;i++){const delta=a[i]-b[i];if(Math.abs(delta)>1e-7)return delta;}return 0;
}
const originKey=(r:RawCandidate)=>[r.position.x,r.position.y,r.position.z,r.size.w,r.size.d,r.size.h,r.orientation].join(':');
export function extraOrigins(input:PlanningInput,deadline:number){
 const raw:RawCandidate[]=[],clock=()=>performance.now();let complete=true;
 for(const orientation of allowedOrientations(input.current.size,input.current.orientationAllowed,input.current.handling,input.current.maxLoadByAxis)){
  if(clock()>=deadline){complete=false;break;}
  const size=oriented(input.current.size,orientation),base=frontierOrigins(input.pallet,size,input.placements,input.constraints.horizontalGap);
  raw.push(...base.map(position=>({position,size,orientation})));

interface DenseRoot { placement:Placement; path:Candidate['path']; rank:number[]; candidate?:Candidate }
function rootCandidate(root:DenseRoot,input:PlanningInput):Candidate{
 if(root.candidate)return root.candidate;

/** Paired sampled scenarios, identical for every root; no actual future arrivals.
 * Root candidates are constrained to the same minimum bottom z and minimum new void.
 * Volume first, count second, then predicted new void and the deterministic dense rank.
 */
function denseLookahead(input:PlanningInput,roots:DenseRoot[],deadline:number,publish:(candidate:Candidate)=>void){
 const depth=Math.min(4,Object.values(input.remaining).reduce((s,n)=>s+n,0)),samples=3;
 if(!depth||roots.length<2)return {candidate:null,completeScenarios:0,simulations:0,interrupted:false};
 const random=rng(parseInt(fingerprint({seed:input.settings.plannerSeed,current:input.current.id,remaining:input.remaining,policy:'dense-volume'}),16));
 const sequences=Array.from({length:samples},(_,sample)=>{
  const counts={...input.remaining},sequence:Observation[]=[];
  for(let at=0;at<depth;at++){
   let draw=random()*Object.values(counts).reduce((s,n)=>s+n,0);const t=input.types.find(t=>(draw-=counts[t.id]??0)<0)!;counts[t.id]--;
   sequence.push({id:`dense-forecast-${sample}-${at}-${t.id}`,typeId:t.id,size:{...t.size},weight:t.weight,status:'normal',orientationAllowed:[...t.orientations],maxLoadKg:t.maxLoadKg,maxLoadSource:t.maxLoadSource,maxLoadByAxis:t.maxLoadByAxis,material:t.material,handling:t.handling,strengthFactor:t.strengthFactor,friction:t.friction});
  }
  return sequence;
 });
 const totals=roots.map(()=>({volume:0,count:0,void:0}));let completeScenarios=0,simulations=0,interrupted=false,candidate:Candidate|null=null;
 for(const sequence of sequences){
  const results:{volume:number;count:number;void:number}[]=[];
  for(const root of roots){
   if(performance.now()>=deadline){interrupted=true;break;}
   let stack=inspectConstraints(root.placement,input.current,input.placements,input.pallet,input.constraints).stack,amount=0,count=0,voidAmount=0;
   const remaining={...input.remaining};let complete=true;
   for(const current of sequence){
    if(performance.now()>=deadline){complete=false;break;}remaining[current.typeId]--;
    const next:PlanningInput={...input,current,remaining:{...remaining},placements:stack,strategyState:undefined,settings:{...input.settings,inventoryMode:'none',maxCandidates:24}};
    const origins=candidateOrigins(next,true).budget;let chosen:ReturnType<typeof inspectConstraints>|null=null,rank:number[]|null=null;
    for(const raw of origins){
     if(performance.now()>=deadline){complete=false;break;}
     if(!geometricFeasible(raw,stack,input.pallet,input.constraints))continue;
     const checked=inspectConstraints(fromOrigin(raw,next),current,stack,input.pallet,input.constraints);if(checked.reasons.length)continue;
     const r=denseRank(checked.placement,next);if(!rank||compareDenseRank(r,rank)<0){chosen=checked;rank=r;}
    }
    if(!complete)break;
    if(chosen){voidAmount+=rank![1];stack=chosen.stack;amount+=current.size.w*current.size.d*current.size.h;count++;}
   }
   if(!complete){interrupted=true;break;}
   simulations++;results.push({volume:amount,count,void:voidAmount});
  }
  // Partial scenarios cannot bias a root merely because it was evaluated first.
  if(interrupted||results.length!==roots.length)break;
  results.forEach((r,i)=>{totals[i].volume+=r.volume;totals[i].count+=r.count;totals[i].void+=r.void;});completeScenarios++;
  const ranked=roots.map((root,i)=>({root,...totals[i]})).sort((a,b)=>b.volume-a.volume||b.count-a.count||a.void-b.void||compareDenseRank(a.root.rank,b.root.rank));
  const next=rootCandidate(ranked[0].root,input);if(performance.now()<deadline){candidate=next;publish(next);}else{interrupted=true;break;}
 }
 return {candidate,completeScenarios,simulations,interrupted};
}
export function searchFloorDense(request:DecisionRequest,baseCandidates:Candidate[],deadline:number,publish:(candidate:Candidate)=>void){
 const input=request.input,seen=new Set<string>();let best:Candidate|null=null,bestRank:number[]|null=null,checked=0,geometricRejected=0,hardRejected=0,dominated=0,generated=0,expired=false,extraComplete=true;
 const rootGroups=new Map<string,DenseRoot>();let rootTop:DenseRoot[]=[],rootPrefix:number[]|null=null;

  const prefix=root.rank.slice(0,2),cmp=rootPrefix?compareDenseRank(prefix,rootPrefix):-1;
  if(cmp>0)return;if(cmp<0){rootGroups.clear();rootTop=[];rootPrefix=prefix;}
  const p=root.placement,q=Number(p.position.x+p.size.w/2>=input.pallet.width/2)+2*Number(p.position.y+p.size.d/2>=input.pallet.depth/2),key=`${q}:${p.orientation}`;
  const old=rootGroups.get(key);if(!old||compareDenseRank(root.rank,old.rank)<0)rootGroups.set(key,root);
  if(!rootTop.some(r=>originKey(r.placement)===originKey(p)))rootTop=[...rootTop,root].sort((a,b)=>compareDenseRank(a.rank,b.rank)).slice(0,2);
 };
 const accept=(candidate:Candidate)=>{const rank=denseRank(candidate.placement,input);if(performance.now()>=deadline){expired=true;return;}remember({placement:candidate.placement,path:candidate.path,rank,candidate});if(!bestRank||compareDenseRank(rank,bestRank)<0){best=candidate;bestRank=rank;publish(candidate);}};
 for(const c of baseCandidates){seen.add(originKey(c.placement));if(c.valid)accept(c);}
 const inspect=(raw:RawCandidate[])=>{
  generated+=raw.length;
  for(const r of raw){
   if(performance.now()>=deadline){expired=true;return;}
   const key=originKey(r);if(seen.has(key))continue;seen.add(key);
   if(bestRank&&r.position.z>bestRank[0]+EPS){dominated++;continue;}

   const result=inspectConstraints(fromOrigin(r,input),input.current,input.placements,input.pallet,input.constraints);checked++;

   const rank=denseRank(result.placement,input);if(performance.now()>=deadline){expired=true;return;}
   const root:DenseRoot={placement:result.placement,path:result.path,rank};remember(root);
   if(bestRank&&compareDenseRank(rank,bestRank)>=0)continue;
   const candidate=rootCandidate(root,input);
   if(performance.now()<deadline){best=candidate;bestRank=rank;publish(candidate);}else{expired=true;return;}
  }
 };
 if(performance.now()<deadline){const expanded={...input,settings:{...input.settings,maxCandidates:128}};inspect([...candidateOrigins(expanded,false).budget,...candidateOrigins(expanded,true).budget]);}
 if(!expired&&performance.now()<deadline){const extra=extraOrigins(input,deadline);extraComplete=extra.complete;inspect(extra.raw);}
 const roots=[...new Map([...rootTop,...[...rootGroups.values()].sort((a,b)=>compareDenseRank(a.rank,b.rank))].map(r=>[originKey(r.placement),r])).values()].slice(0,6);
 let future:ReturnType<typeof denseLookahead>={candidate:null,completeScenarios:0,simulations:0,interrupted:false};
 if(best&&!expired&&performance.now()<deadline)future=denseLookahead(input,roots,deadline,publish);

 if(performance.now()>=deadline)expired=true;
 return {candidate:best,simulations:future.simulations,completeScenarios:future.completeScenarios,diagnostics:{policy:DENSE_POLICY_VERSION,priority:DENSE_PRIORITY.join(' → '),rank:best?denseRank(best.placement,input):[],rootCandidates:baseCandidates.length,generatedOrigins:generated,uniqueOrigins:seen.size,fullConstraintChecks:checked,geometricRejected,hardRejected,lexicographicallyDominated:dominated,
  forecastRootCount:roots.length,forecastDepth:4,forecastScenarioCap:3,forecastCompleteScenarios:future.completeScenarios,forecastSimulations:future.simulations,forecastObjective:'same lowest z and least new void, then paired predicted added volume, count, predicted new void, dense rank',
  stop:expired?'search_deadline':extraComplete&&!future.interrupted?'enumerated_origins_and_forecasts_exhausted':'partial_search',candidatesComplete:extraComplete&&!expired?1:0}};
}

/** Floor first and exact new buried void first for all three policies.
 * Only the tie-breaking objective changes. Deliberately dispersed is an artificial
 * negative control; it still obeys every shared physical constraint.
 */
export function clusterRank(box:Placement,input:PlanningInput,policy:DecisionRequest['algorithm']){
 const dense=denseRank(box,input);let same=0,zone=0,total=0;
 for(const other of input.placements){const c=contact(box,other);total+=c.area;if(box.typeId===other.typeId)same+=c.area;if(department(box.typeId)===department(other.typeId))zone+=c.area;}
 const norm=2*(box.size.w*box.size.d+box.size.w*box.size.h+box.size.d*box.size.h);
 const purity=same/norm,zonePurity=zone/norm;
 // Layer policy allows same-SKU preference only inside a 250mm bottom-height band.
 // Connected policy considers every valid upper-layer height after filling the floor.
 const layer=policy==='cluster-layer'?0:Math.floor(box.position.z/250);
 return [box.position.z<1e-5?0:1,dense[1],layer,policy==='dispersed'?purity:-purity,
  policy==='dispersed'?zonePurity:-zonePurity,dense[0],dense[2],-total,...dense.slice(3)];
}

### src/engine/mcts.ts
import { candidateSet } from './vendor/pallet/candidates';

import { fingerprint } from './vendor/pallet/benchmark/statistics';
import { top, volume } from './vendor/pallet/geometry';
import type { Candidate, Observation, Placement, PlanningInput } from './types';
import type { DecisionRequest } from './types';
interface State { placed: Placement[]; count: number; vol: number }
interface Node { action: Candidate | null; visits: number; sum: number; countSum: number; volumeSum: number; children: Node[] | null }
const node = (action: Candidate | null): Node => ({ action, visits: 0, sum: 0, countSum: 0, volumeSum: 0, children: null });
/** Logical port of alps-control-room engine/search.js: scenario UCT, expansion, stochastic rollout, backpropagation.
 * Geometry, root generation and hard constraints come exclusively from this app's canonical TS engine.
 * It is a new common-engine adapter, not a claim of bit-identical outputs from the old engine.
 */
export function searchMcts(request: DecisionRequest, rootCandidates: Candidate[], deadline: number, publish: (c: Candidate) => void) {
  const input = request.input, clock = () => performance.now(), roots = rootCandidates.filter(c => c.valid).sort((a,b) => b.score-a.score || a.id.localeCompare(b.id)).slice(0, 6);
  if (!roots.length) return { candidate: null, simulations: 0, completeScenarios: 0, fallback: false };
  const depth = Math.min(request.mcts.depth, Object.values(input.remaining).reduce((s,v) => s+v,0));
  if (!depth) return { candidate: roots[0], simulations: 0, completeScenarios: 0, fallback: false };
  const random = rng(parseInt(fingerprint({ seed: input.settings.plannerSeed, current: input.current.id, remaining: input.remaining }),16));
  const sample = () => {
    const counts = { ...input.remaining }, sequence: Observation[] = [];
    for (let at = 0; at < depth; at++) {
      const total = Object.values(counts).reduce((s,v) => s+v,0); if (!total) break;
      let draw = random()*total;
      const type = input.types.find(t => (draw -= counts[t.id] ?? 0) < 0)!;
      counts[type.id]--;
      sequence.push({ id:`forecast-${at}-${type.id}`, typeId:type.id, size:{...type.size}, weight:type.weight,
        status:'normal', orientationAllowed:[...type.orientations], maxLoadKg:type.maxLoadKg,maxLoadSource:type.maxLoadSource,
        maxLoadByAxis:type.maxLoadByAxis, material:type.material,handling:type.handling,friction:type.friction,strengthFactor:type.strengthFactor });
    }
    return sequence;
  };
  const sequences = Array.from({length:request.mcts.scenarios},sample);
  const forests = sequences.map(() => ({...node(null),children:roots.map(c=>node(c))}));
  let simulations=0;
  const apply = (state:State,c:Candidate|null):State => {
    if (!c) return state;
    const source = c.placement.id === input.current.id ? input.current : sequences.flat().find(o=>o.id===c.placement.id)!;
    return {placed:inspectConstraints(c.placement,source,state.placed,input.pallet,input.constraints).stack,count:state.count+1,vol:state.vol+volume(c.placement.size)};
  };
  const future = (state:State,seq:Observation[],index:number) => {
    const remaining={...input.remaining}; for(let j=0;j<=index;j++) remaining[seq[j].typeId]--;
    const result=candidateSet({...input,current:seq[index],remaining,placements:state.placed,strategyState:undefined,
      settings:{...input.settings,inventoryMode:'none',maxCandidates:Math.min(28,input.settings.maxCandidates)}});
    return result.candidates.filter(c=>c.valid).sort((a,b)=>b.score-a.score||a.id.localeCompare(b.id)).slice(0,3);
  };
  const rollout=(initial:State,seq:Observation[],index:number):State|null=>{
    let state=initial;
    for(let i=index;i<seq.length;i++){
      if(clock()>=deadline)return null;
      const choices=future(state,seq,i); if(clock()>=deadline)return null;
      if(choices.length)state=apply(state,choices[random()<.15?Math.floor(random()*choices.length):0]);
    }
    return state;
  };
  const choose=()=>{
    const complete=forests.filter(f=>f.children.every(c=>c.visits>0));
    if(!complete.length)return {candidate:null,completeScenarios:0};
    const ranked=roots.map((c,i)=>({c,count:complete.reduce((s,f)=>s+f.children[i].countSum/f.children[i].visits,0)/complete.length,
      vol:complete.reduce((s,f)=>s+f.children[i].volumeSum/f.children[i].visits,0)/complete.length}));
    ranked.sort((a,b)=>b.count-a.count||b.vol-a.vol||b.c.score-a.c.score);
    return {candidate:ranked[0].c,completeScenarios:complete.length};
  };
  while(clock()<deadline&&simulations<request.mcts.maxIterations){
    const warming=forests.findIndex(f=>f.children.some(c=>!c.visits)),si=warming>=0?warming:simulations%sequences.length,seq=sequences[si],tree=forests[si];
    const trail:Node[]=[tree];let current:Node=tree,at=0,state:State={placed:input.placements,count:0,vol:0},complete=false;
    while(clock()<deadline){
      const children=current.children!;
      const child=children.find(c=>!c.visits)??children.reduce((a,b)=>{
        const u=(n:Node)=>n.sum/n.visits+Math.SQRT2*Math.sqrt(Math.log(Math.max(1,current.visits))/n.visits);return u(a)>=u(b)?a:b;
      });
      trail.push(child);state=apply(state,child.action);
      if(at>=seq.length){complete=true;break;}
      if(child.children===null){
        const choices=future(state,seq,at);if(clock()>=deadline)break;
        child.children=[...choices,null].map(node);const expansion=child.children[0];trail.push(expansion);state=apply(state,expansion.action);
        const end=rollout(state,seq,at+1);if(end){state=end;complete=true;}break;
      }
      current=child;at++;
    }
    if(!complete)break;
    const possible=volume(input.current.size)+seq.reduce((s,o)=>s+volume(o.size),0),quality=1/(1+Math.max(0,...state.placed.map(top)));
    // Original port's count-first reward and lexicographic root comparison; not a reported combined quality score.
    const reward=state.count/(depth+1)+.0001*state.vol/Math.max(1,possible)+.000001*quality;
    for(const n of trail){n.visits++;n.sum+=reward;n.countSum+=state.count;n.volumeSum+=state.vol;}
    simulations++;
    if(simulations%roots.length===0){const best=choose();if(best.candidate&&clock()<deadline)publish(best.candidate);}
  }
  const best=choose();return {...best,simulations,fallback:!best.candidate};
}

/** Sequential, rule-based handling model for the actual final pallet coordinates.
 * It is not a store walking model or a robot reach/collision certification.
 * Every algorithm uses the same zone order, top-access rule and movement duration.
 */
export type RedistributionZone = 0 | 1 | 2;
export interface HandlingPoint { x: number; y: number; z: number }
export interface RedistributionBox {
  id: string; typeId: string; position: HandlingPoint;
  size: { w: number; d: number; h: number }; weight: number; color?: string;
}
export type RedistributionStage = 'pallet-to-buffer' | 'pallet-to-destination' | 'buffer-to-destination';
export interface RedistributionStep {
  index: number; boxId: string; box: RedistributionBox; zone: RedistributionZone;
  stage: RedistributionStage; from: HandlingPoint; to: HandlingPoint;
  fromSlot: number | null; toSlot: number | null;
  start: number; end: number; seconds: number;
}
export interface RedistributionOptions {
  cycleSeconds: 10 | 12;
  palletWidthMm?: number; palletDepthMm?: number; unplacedCount?: number;
  zoneOrder?: readonly RedistributionZone[];
}
export interface RedistributionPlan {
  schema: 'alps-redistribution/1'; initialBoxes: RedistributionBox[];
  zoneOrder: RedistributionZone[]; cycleSeconds: 10 | 12;
  steps: RedistributionStep[];
  bufferSlots: { slot: number; position: HandlingPoint }[];
  destinations: { boxId: string; zone: RedistributionZone; slot: number; position: HandlingPoint }[];
  finalBoxes: RedistributionBox[];
  metrics: {
    targetBoxes: number; excludedUnplaced: number; extraHandles: number;
    totalMoves: number; totalSeconds: number; secondsPer100Boxes: number | null;
    bufferPeak: number; deliveredBoxes: number; conserved: boolean;
  };
  assumptions: string[];
}

const ZONES: Record<string, RedistributionZone> = { S1: 0, S2: 0, S3: 1, S4: 1, S5: 2, S6: 2 };
const EPS = 1e-6;
const top = (b: RedistributionBox) => b.position.z + b.size.h;
const overlapXY = (a: RedistributionBox, b: RedistributionBox) =>
  a.position.x < b.position.x + b.size.w - EPS && b.position.x < a.position.x + a.size.w - EPS
  && a.position.y < b.position.y + b.size.d - EPS && b.position.y < a.position.y + a.size.d - EPS;
const intersect = (a: RedistributionBox, b: RedistributionBox) => overlapXY(a, b)
  && a.position.z < top(b) - EPS && b.position.z < top(a) - EPS;
const byTop = (a: RedistributionBox, b: RedistributionBox) => top(b) - top(a) || a.id.localeCompare(b.id);
export function redistributionZone(typeId: string): RedistributionZone {
  const zone = ZONES[typeId];
  if (zone === undefined) throw Error(`No redistribution zone for SKU: ${typeId}`);
  return zone;
}
/** All vertically higher boxes sharing the footprint obstruct a vertical extraction. */
export function redistributionBlockers(box: RedistributionBox, boxes: readonly RedistributionBox[]): RedistributionBox[] {
  return boxes.filter(other => other.id !== box.id && other.position.z >= top(box) - EPS && overlapXY(box, other)).sort(byTop);
}

export function planRedistribution(input: readonly RedistributionBox[], options: RedistributionOptions): RedistributionPlan {
  const cycle = options.cycleSeconds, width = options.palletWidthMm ?? 1200, depth = options.palletDepthMm ?? 1000;
  const excluded = options.unplacedCount ?? 0, order: RedistributionZone[] = [...(options.zoneOrder ?? [0, 1, 2])];
  if (![10, 12].includes(cycle)) throw Error('Redistribution cycle must be 10 or 12 seconds');
  if (![width, depth].every(n => Number.isFinite(n) && n > 0)) throw Error('Invalid pallet dimensions');
  if (!Number.isInteger(excluded) || excluded < 0) throw Error('Invalid unplaced count');
  if (order.length !== 3 || new Set(order).size !== 3 || order.some(z => ![0, 1, 2].includes(z))) throw Error('Zone order must contain zones 0, 1 and 2 exactly once');
  if (new Set(input.map(b => b.id)).size !== input.length) throw Error('Duplicate box identity');
  for (const b of input) {
    redistributionZone(b.typeId);
    if (!b.id || ![...Object.values(b.position), ...Object.values(b.size), b.weight].every(Number.isFinite)
      || Object.values(b.size).some(n => n <= 0) || Object.values(b.position).some(n => n < 0) || b.weight < 0) throw Error(`Invalid box geometry: ${b.id}`);
    if (b.position.x + b.size.w > width + EPS || b.position.y + b.size.d > depth + EPS) throw Error(`Box outside pallet: ${b.id}`);
  }
  for (let i = 0; i < input.length; i++) if (input.slice(i + 1).some(b => intersect(input[i], b))) throw Error(`Overlapping input boxes: ${input[i].id}`);
  const initialBoxes = structuredClone([...input]), pallet = new Map(initialBoxes.map(b => [b.id, b]));
  const buffer = new Map<string, { box: RedistributionBox; slot: number; position: HandlingPoint }>();
  const delivered = new Map<string, RedistributionBox>(), occupiedSlots = new Set<number>();
  const bufferSlots = new Map<number, HandlingPoint>(), destinationCounts = [0, 0, 0];
  const steps: RedistributionStep[] = [], destinations: RedistributionPlan['destinations'] = [];
  const pitchX = Math.max(100, ...initialBoxes.map(b => b.size.w)) + 100;
  const pitchY = Math.max(100, ...initialBoxes.map(b => b.size.d)) + 100;
  let bufferPeak = 0;
  const destinationFor = (zone: RedistributionZone) => {
    const slot = destinationCounts[zone]++;
    // Nonoverlapping staging cells are explanatory geometry, not a robot workspace claim.
    return { slot, position: { x: width + 400 + zone * 3 * pitchX + (slot % 3) * pitchX,
      y: Math.floor(slot / 3) * pitchY, z: 0 } };
  };
  const add = (box: RedistributionBox, stage: RedistributionStage, from: HandlingPoint, to: HandlingPoint,
    fromSlot: number | null, toSlot: number | null) => {
    const start = steps.length * cycle;
    steps.push({ index: steps.length, boxId: box.id, box: structuredClone(box), zone: redistributionZone(box.typeId),
      stage, from: { ...from }, to: { ...to }, fromSlot, toSlot, start, end: start + cycle, seconds: cycle });
  };
  const deliver = (box: RedistributionBox, from: HandlingPoint, fromSlot: number | null) => {
    if (delivered.has(box.id)) throw Error(`Box delivered twice: ${box.id}`);
    const zone = redistributionZone(box.typeId), target = destinationFor(zone);
    add(box, fromSlot === null ? 'pallet-to-destination' : 'buffer-to-destination', from, target.position, fromSlot, target.slot);
    if (fromSlot === null) pallet.delete(box.id);
    else { buffer.delete(box.id); occupiedSlots.delete(fromSlot); }
    destinations.push({ boxId: box.id, zone, ...target });
    delivered.set(box.id, { ...structuredClone(box), position: { ...target.position } });
  };
  const hold = (box: RedistributionBox) => {
    let slot = 0; while (occupiedSlots.has(slot)) slot++;
    const position = { x: -400 - (slot % 3 + 1) * pitchX, y: Math.floor(slot / 3) * pitchY, z: 0 };
    add(box, 'pallet-to-buffer', box.position, position, null, slot);
    pallet.delete(box.id); occupiedSlots.add(slot); bufferSlots.set(slot, position);
    buffer.set(box.id, { box, slot, position }); bufferPeak = Math.max(bufferPeak, buffer.size);
  };
  const extract = (box: RedistributionBox, currentZone: RedistributionZone) => {
    if (!pallet.has(box.id)) return;
    for (const blocker of redistributionBlockers(box, [...pallet.values()])) extract(blocker, currentZone);
    if (redistributionBlockers(box, [...pallet.values()]).length) throw Error(`Blocked vertical extraction: ${box.id}`);
    // A target-zone blocker can go directly to its destination; no artificial baseline penalty.
    if (redistributionZone(box.typeId) === currentZone) deliver(box, box.position, null); else hold(box);
  };
  for (const zone of order) {
    const wanted = initialBoxes.filter(b => redistributionZone(b.typeId) === zone).sort(byTop);
    for (const box of wanted) {
      if (delivered.has(box.id)) continue;
      const held = buffer.get(box.id);
      if (held) deliver(held.box, held.position, held.slot); else extract(box, zone);
    }
  }
  const extraHandles = steps.filter(s => s.stage === 'pallet-to-buffer').length;
  const conserved = delivered.size === initialBoxes.length && !pallet.size && !buffer.size && !occupiedSlots.size
    && destinations.length === initialBoxes.length && steps.length === initialBoxes.length + extraHandles;
  if (!conserved) throw Error('Redistribution cargo conservation failed');
  return { schema: 'alps-redistribution/1', initialBoxes, zoneOrder: order, cycleSeconds: cycle, steps,
    bufferSlots: [...bufferSlots].map(([slot, position]) => ({ slot, position })), destinations,
    finalBoxes: [...delivered.values()],
    metrics: { targetBoxes: initialBoxes.length, excludedUnplaced: excluded, extraHandles,
      totalMoves: steps.length, totalSeconds: steps.length * cycle,
      secondsPer100Boxes: initialBoxes.length ? steps.length * cycle / initialBoxes.length * 100 : null,
      bufferPeak, deliveredBoxes: delivered.size, conserved },
    assumptions: [
      'Only boxes actually placed on the source pallet are redistributed; unplaced cargo is excluded and counted separately.',
      'All policies use the same zone order and top-access handling rule. A target-zone blocker is delivered directly; other-zone blockers use a temporary slot.',
      'Every pallet-to-zone, pallet-to-buffer and buffer-to-zone movement uses the same assumed 10/12 second cycle.',
      'Buffer slots and destination cells are nonoverlapping explanatory staging locations. Space, grasp, reach, full-link collision and transport stability at these locations are not validated.',
      'This is sequential handling model time, not measured human work, store walking, shelf stocking or real robot cycle time.',
      'Compare equal target cargo where possible; seconds per 100 boxes is a normalization, not a correction for different SKU mixes or causal proof.',
    ] };
}

export interface RedistributionFrame {
  pallet: RedistributionBox[]; buffer: RedistributionBox[]; destination: RedistributionBox[];
  moving: { step: RedistributionStep; box: RedistributionBox; progress: number } | null;
  completedMoves: number; deliveredBoxes: number; done: boolean;
}
/** State projection used by replay. A carried box belongs to exactly one moving state. */
export function redistributionFrame(plan: RedistributionPlan, seconds: number): RedistributionFrame {
  if (!Number.isFinite(seconds)) throw Error('Replay time must be finite');
  const time = Math.max(0, Math.min(seconds, plan.metrics.totalSeconds));
  const pallet = new Map(plan.initialBoxes.map(b => [b.id, structuredClone(b)]));
  const buffer = new Map<string, RedistributionBox>(), destination = new Map<string, RedistributionBox>();
  let moving: RedistributionFrame['moving'] = null, completedMoves = 0;
  for (const step of plan.steps) {
    if (time < step.start) break;
    const source = step.stage === 'buffer-to-destination' ? buffer : pallet;
    const box = source.get(step.boxId);
    if (!box) throw Error(`Replay source box missing: ${step.boxId}`);
    source.delete(step.boxId);
    if (time < step.end) {
      moving = { step, box: { ...box, position: { ...step.from } }, progress: (time - step.start) / step.seconds };
      break;
    }
    const target = step.stage === 'pallet-to-buffer' ? buffer : destination;
    target.set(step.boxId, { ...box, position: { ...step.to } }); completedMoves++;
  }
  return { pallet: [...pallet.values()], buffer: [...buffer.values()], destination: [...destination.values()],
    moving, completedMoves, deliveredBoxes: destination.size, done: completedMoves === plan.steps.length };
}
