/**
 * Environment-side events. `at` is the 0-based arrival index at which the event becomes
 * observable; nothing about an event is given to the planner before that arrival.
 * - missing: the box at `at` never arrives; its type count is corrected at that moment.
 * - damaged: the box arrives damaged and must be isolated, not placed.
 * - resize: the box arrives with corrected size/weight (catalogue was wrong); observed correctly.
 * - mismeasure: the first observation is wrong; the true size is measured at pick, before placing.
 * - buffer-down: the staging buffer accepts no new box for `duration` arrivals.
 * - disturb: the top box of the current pallet is found displaced by `shiftMm`; the stack must be
 *   re-verified and, if invalid, quarantined instead of loaded further.
 */
export type EventKind='missing'|'damaged'|'resize'|'mismeasure'|'buffer-down'|'disturb';
export interface HarnessEvent { kind:EventKind;at:number;size?:Dimensions;weight?:number;duration?:number;shiftMm?:number }
export interface CaseSource { dataset:'synthetic'|'repeated-sku'|'bed-bpp';orderIds:string[];generator:string }

export interface CaseInput { situation:string;phase:Phase;seeds:CaseSeeds;scenario:Scenario;sequence:string[];events:HarnessEvent[];bufferSize:number;source:CaseSource }

export interface EventProfile { kinds:EventKind[];count:number;sizeChange:[number,number];weightChange:[number,number] }
export interface Situation { id:string;label:string;question:string;bufferSize:number;events?:EventProfile;build:(phase:Phase,seeds:CaseSeeds)=>CaseInput }

function cell(s:Scenario):Scenario{
 s.supplyMode='arrival';s.events=[];
 s.constraints={...s.constraints,robotMode:'gripper',heavyRule:'each',reach:undefined,workspace:{...s.constraints.workspace,xMin:-5000,xMax:5000,yMin:-5000,yMax:5000},stability:{...s.constraints.stability!}};
 return s;
}
type Order='shuffle'|'large-late'|'heavy-late'|'source';
/** Arrival sequence from the catalogue; the planner never receives it. */
function arrival(types:BoxType[],order:Order,seed:number):string[]{
 const pool=types.flatMap(t=>Array.from({length:t.quantity},()=>t.id)),byId=new Map(types.map(t=>[t.id,t]));
 const q=shuffled(pool,rng(seed));
 if(order==='large-late')q.sort((a,b)=>volume(byId.get(a)!.size)-volume(byId.get(b)!.size));
 if(order==='heavy-late')q.sort((a,b)=>byId.get(a)!.weight-byId.get(b)!.weight);
 return q;
}
function events(profile:EventProfile|undefined,s:Scenario,sequence:string[],seed:number):HarnessEvent[]{
 if(!profile)return [];
 const random=rng(seed),n=sequence.length,used=new Set<number>(),list:HarnessEvent[]=[],between=([a,b]:[number,number])=>a+random()*(b-a);
 for(let i=0;i<profile.count;i++){
  let at=0;do{at=Math.floor(n*.2+random()*n*.6);}while(used.has(at));used.add(at);
  const kind=profile.kinds[i%profile.kinds.length],t=s.types.find(t=>t.id===sequence[at])!;
  const scale=(d:Dimensions)=>({w:Math.round(d.w*(1+between(profile.sizeChange))),d:Math.round(d.d*(1+between(profile.sizeChange))),h:Math.round(d.h*(1+between(profile.sizeChange)))});
  if(kind==='resize')list.push({kind,at,size:scale(t.size),weight:Math.round(t.weight*(1+between(profile.weightChange))*100)/100});
  else if(kind==='mismeasure')list.push({kind,at,size:scale(t.size)});
  else if(kind==='buffer-down')list.push({kind,at,duration:5});
  else if(kind==='disturb')list.push({kind,at,shiftMm:Math.round(between([40,120]))});
  else list.push({kind,at});
 }
 return list.sort((a,b)=>a.at-b.at);
}
type Build=(phase:Phase,seeds:CaseSeeds)=>Omit<CaseInput,'situation'>;
function synthetic(count:number,order:Order,bufferSize=2,profile?:EventProfile,support?:number):Build{
 return (phase,seeds)=>{

  const sequence=arrival(s.types,order,seeds.arrival);
  return {phase,seeds,scenario:s,sequence,events:events(profile,s,sequence,seeds.event),bufferSize,source:{dataset:'synthetic',orderIds:[],generator:`streamInventory(${seeds.dataset},${count})`}};
 };
}

  s.id='repeated-sku';s.name='반복 규격 5종 60개 · KS T11';s.types=input.types;s.generation=input.generation;s.pallet={width:1100,depth:1100,maxHeight:1500};
  const sequence=arrival(s.types,order,seeds.arrival);
  return {phase,seeds,scenario:s,sequence,events:[],bufferSize:2,source:{dataset:'repeated-sku',orderIds:[],generator:`heterogeneousInventory(${seeds.dataset},5,60,repeated)`}};
 };
}
/** BED-BPP from the phase's own order pool only. `count` undefined = one complete order. */
function bed(order:Order,count?:number):Build{
 return (phase,seeds)=>{
  const pool=BED_SPLIT[phase],start=seeds.dataset%pool.length,orderIds:string[]=[],items:ReturnType<typeof bedOrderItems>=[];
  for(let k=0;k<pool.length&&(count===undefined?orderIds.length<1:items.length<count);k++){const id=pool[(start+k)%pool.length];orderIds.push(id);items.push(...bedOrderItems(id));}
  const chosen=count===undefined?items:items.slice(0,count),s=cell(streamInventory(seeds.dataset,8));
  s.id='bed-bpp';s.name=`BED-BPP ${orderIds.join('+')}`;s.pallet={width:1200,depth:800,maxHeight:1600};
  s.constraints.standingHeight={enabled:false,maxRiseMm:100};s.constraints.stability={...s.constraints.stability!,slendernessMode:'score'};
  // One catalogue entry per distinct (size, weight); the source sequence stays environment-side.
  const key=(i:typeof chosen[number])=>`${i.size.w}x${i.size.d}x${i.size.h}@${i.weight}`,types=new Map<string,BoxType>();
  for(const item of chosen){const k=key(item),t=types.get(k);if(t)t.quantity++;else types.set(k,{id:k,name:k,size:{...item.size},weight:item.weight,quantity:1,orientations:[0,90],material:'unknown',color:'#a9c4b0'});}
  s.types=[...types.values()];
  const sequence=order==='source'?chosen.map(key):arrival(s.types,order,seeds.arrival);
  return {phase,seeds,scenario:s,sequence,events:[],bufferSize:2,source:{dataset:'bed-bpp',orderIds,generator:`BED-BPP v1 subset · ${count??'complete order'} · ${order}`}};
 };
}
const exceptions=(kinds:EventKind[],count:number):EventProfile=>({kinds,count,sizeChange:[-.12,.12],weightChange:[-.3,.3]});

/**
 * Independent re-verification of a final stack: geometry without the planner checks, then a
 * rebuild that re-inspects every box in commit order against the boxes before it.
 */
export function auditStack(stack:readonly Placement[],s:Scenario,observed:Map<string,Observation>){
 const p=s.pallet,violations:Record<string,number>={},add=(k:string)=>violations[k]=(violations[k]??0)+1;
 for(const b of stack){if(b.position.x< -EPS||b.position.y< -EPS||b.position.z< -EPS||b.position.x+b.size.w>p.width+EPS||b.position.y+b.size.d>p.depth+EPS||top(b)>p.maxHeight+EPS)add('geometry');}
 for(let i=0;i<stack.length;i++)for(let j=i+1;j<stack.length;j++)if(intersects(stack[i],stack[j],s.constraints.horizontalGap))add('geometry');
 let rebuilt:Placement[]=[];const c={...s.constraints,robotMode:'ideal' as const};
 for(const b of stack){const o=observed.get(b.id);if(!o){add('other');continue;}
  const r=inspectConstraints({...b,supports:[],supportRatio:0,loadAbove:0},o,rebuilt,p,c);for(const reason of new Set(r.reasons.map(category)))add(reason);rebuilt=r.stack;}
 const loads=loadSummary([...stack],s.constraints.stability),cats=['geometry','support','heavy-rule','load','local-stability','slenderness','standing-height'];
 const checks:Record<string,Status>=Object.fromEntries(cats.map(k=>[k,violations[k]?'fail':'pass']));
 if(!violations.load&&loads.unknownCapacity>0)checks.load='unverified';
 Object.assign(checks,{'robot-path-approximate':'pass','robot-joint-ik':'unverified','measured-compression-strength':'unverified','dynamic-transport':'unverified'});
 return {violations,checks,rebuilt,loads};
}

function choose(policy:SingleRobotPolicy,input:PlanningInput){
 const start=performance.now();
 if(policy.algorithm==='compact'){const {candidates,inspected}=rankCompactSites(input,{smallFloorPenalty:policy.smallFloorPenalty});
  const rejections:Record<string,number>={};if(!candidates.length)for(const c of inspected)for(const r of c.reasons)rejections[category(r)]=(rejections[category(r)]??0)+1;
  return {candidate:candidates[0] as Candidate|undefined,analysis:null as Analysis|null,rejections,ms:performance.now()-start};}
 const analysis=plan(input),candidate=analysis.candidates.find(c=>c.id===analysis.selectedId&&c.valid),rejections:Record<string,number>={};
 if(!candidate)for(const [r,n] of Object.entries(analysis.rejections))rejections[category(r)]=(rejections[category(r)]??0)+n;
 return {candidate,analysis,rejections,ms:performance.now()-start};
}

/**

 * reveals each only at its arrival index. Planners see an order-free catalogue, the current box,

 * When the buffer is full the pallet is closed and the box retried on an empty pallet; a box that

function palletRecord(s:Scenario,placed:Placement[],index:number,closedBy:PalletRecord['closedBy'],observed:Map<string,Observation>,gs:number[],unit?:PalletUnitModel):PalletRecord{

  comOffset:center?comOffsetRatio(center,p):null,comHeightMm:center?center.z:null,minSupportRatio:raised.length?Math.min(...raised.map(b=>b.supportRatio)):1,
  minLocalReserveMm:stability(placed,s.constraints.stability).minReserve,loadUtilization:audit.loads.loadUtilization,unknownCapacity:audit.loads.unknownCapacity,
  transport:{loose,wrapped,palletUnit:unit?pu:null},slidingBoxes:sliding,audit:{violations:audit.violations,checks:audit.checks}};
}

import { massCenter } from './packaging/spec';
import { convexHull,EPS } from './geometry';
import { stability } from './stability';
import { DEFAULT_STABILITY } from './types';
import type { Pallet,Placement,Vec3,BoxType,StabilitySettings } from './types';

/** Whole-load centre of mass on the pallet. Box CoG offsets come from BoxSpec when present. */
export function loadCom(placed:readonly Pick<Placement,'position'|'size'|'orientation'|'packaging'|'weight'>[]):{mass:number;center:Vec3|null}{
 const mass=placed.reduce((n,b)=>n+b.weight,0);if(!(mass>0))return {mass:0,center:null};
 const c={x:0,y:0,z:0};for(const b of placed){const m=massCenter(b);c.x+=m.x*b.weight;c.y+=m.y*b.weight;c.z+=m.z*b.weight;}
 return {mass,center:{x:c.x/mass,y:c.y/mass,z:c.z/mass}};
}
/** Larger axis offset from the pallet centre, as a fraction of that half dimension (0 centre, 1 edge). */
export function comOffsetRatio(center:Vec3,p:Pallet){return Math.max(Math.abs(center.x-p.width/2)/(p.width/2),Math.abs(center.y-p.depth/2)/(p.depth/2));}
/** Mean known box mass. Uses declared types and quantities only, never the arrival order. */
export function meanTypeMass(types:readonly BoxType[]){const n=types.reduce((s,t)=>s+t.quantity,0);return n?types.reduce((s,t)=>s+t.weight*t.quantity,0)/n:0;}
/**
 * Soft whole-load balance cost after a placement: offset ratio x mass ramp.
 * The ramp M/(M+4m) keeps the first few light boxes from dominating the decision and
 * approaches 1 as the load grows (m = mean known box mass, or the placed mean when unknown).
 */
export function comCost(placed:readonly Placement[],p:Pallet,meanMass?:number){
 const {mass,center}=loadCom(placed);if(!center)return {offset:0,ramp:0,cost:0};
 const m=meanMass&&meanMass>0?meanMass:mass/placed.length,offset=comOffsetRatio(center,p),ramp=mass/(mass+4*m);
 return {offset,ramp,cost:offset*ramp};
}
/**
 * Transport tipping margins under a horizontal acceleration a (in g), two cargo models:
 * - loose: no wrapping. At every contact plane, the resultant of the box and all transmitted
 *   upper loads must stay inside that plane's support hull by h·a (h = resultant height above
 *   the plane). Uses the same equilibrium/stability model as placement checks.
 * - wrapped: the load moves as one rigid block on the hull of its floor contacts.
 * Neither model includes the pallet itself; see palletUnitMargin.
 */
export function transportMargin(placed:readonly Placement[],p:Pallet,accelerationG:number,base:StabilitySettings=DEFAULT_STABILITY){
 if(!placed.length)return {looseMarginMm:Infinity,wrappedMarginMm:Infinity};
 const loose=stability([...placed],{...base,minMarginRatio:0,lateralAccelerationG:accelerationG}).minReserve;
 const {center}=loadCom(placed),hull=convexHull(placed.filter(b=>Math.abs(b.position.z)<=EPS).flatMap(b=>corners(b)));
 const wrapped=center&&hull.length>=3?edgeMargin(hull,center)-center.z*accelerationG:-Infinity;
 return {looseMarginMm:loose,wrappedMarginMm:wrapped};
}
/** Explicit coupling model for the loaded pallet as one unit on its own footprint. */
export interface PalletUnitModel { palletMassKg:number;deckHeightMm:number;source:string }
/**
 * Loaded-pallet tipping on the pallet footprint. Valid only when the cargo is coupled to the
 * pallet (e.g. wrapped to the deck): the caller must supply the pallet mass, deck height and the
 * source of that assumption. Returns null without a model so it is reported as unverified.
 */
export function palletUnitMargin(placed:readonly Placement[],p:Pallet,accelerationG:number,model?:PalletUnitModel){
 if(!model)return null;
